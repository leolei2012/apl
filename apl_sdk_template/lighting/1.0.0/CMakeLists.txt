cmake_minimum_required(VERSION 3.16)
project(product_light LANGUAGES C)
add_library(product_light STATIC lib/light.c)
target_include_directories(product_light PUBLIC api settings)
target_compile_features(product_light PUBLIC c_std_11)
add_library(light_port_example OBJECT board/light_port.c samples/app.c)
target_include_directories(light_port_example PRIVATE board)
target_link_libraries(light_port_example PRIVATE product_light)
include(CTest)
if(BUILD_TESTING)
  add_executable(product_light_check checks/verify.c)
  target_link_libraries(product_light_check PRIVATE product_light)
  add_test(NAME product_light_check COMMAND product_light_check)
endif()
